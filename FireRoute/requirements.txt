

from pathlib import Path
import pickle

import networkx as nx
import osmnx as ox


# Ruta donde se guardará el mapa
MAP_PATH = Path(__file__).resolve().parent.parent / "data" / "trujillo_graph.pkl"


def build_trujillo_graph():
    """
    Construye el grafo dirigido de Trujillo
    alrededor de la Compañía de Bomberos B-26.
    """

    print("Descargando mapa de Trujillo...")

    # Dirección de referencia de la compañía
    direccion = (
        "Av. España 506, Trujillo, La Libertad, Peru"
    )

    # Descargar calles para vehículos
    G = ox.graph.graph_from_address(
        direccion,
        dist=2000,
        network_type="drive",
        simplify=True
    )

    # Agregar velocidades y tiempos
    G = ox.routing.add_edge_speeds(
        G,
        fallback=30
    )

    G = ox.routing.add_edge_travel_times(G)

    # Crear identificadores consecutivos
    mapping = {
        nodo: i
        for i, nodo in enumerate(G.nodes)
    }

    # Guardar coordenadas de cada nodo
    coords = {
        mapping[nodo]: (
            float(G.nodes[nodo]["y"]),
            float(G.nodes[nodo]["x"])
        )
        for nodo in G.nodes
    }

    # Crear grafo dirigido
    graph = nx.DiGraph()
    graph.add_nodes_from(range(len(mapping)))

    vmax_ms = 0.0

    for u, v, datos in G.edges(data=True):

        origen = mapping[u]
        destino = mapping[v]

        longitud = float(datos["length"])
        velocidad = float(datos["speed_kph"]) / 3.6

        # Tiempo de recorrido en segundos
        costo = longitud / velocidad

        vmax_ms = max(vmax_ms, velocidad)

        # Conservar el trayecto más rápido
        if (
            not graph.has_edge(origen, destino)
            or costo < graph[origen][destino]["weight"]
        ):
            graph.add_edge(
                origen,
                destino,
                weight=costo,
                length=longitud,
                speed_ms=velocidad
            )

    print("Mapa construido correctamente.")

    return graph, coords, vmax_ms


def save_graph(graph, coords, vmax_ms):
    """Guarda el mapa en la carpeta data."""

    MAP_PATH.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    datos = {
        "graph": graph,
        "coords": coords,
        "vmax_ms": vmax_ms
    }

    with open(MAP_PATH, "wb") as archivo:
        pickle.dump(datos, archivo)

    print("Mapa guardado correctamente.")


def load_graph():
    """Carga el mapa guardado anteriormente."""

    with open(MAP_PATH, "rb") as archivo:
        datos = pickle.load(archivo)

    return (
        datos["graph"],
        datos["coords"],
        datos["vmax_ms"]
    )
